\usepackage{titlesec}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Custom section format
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\titleformat{\section}
  {\normalfont\Large\bfseries} % format applied to whole title
  {}                          % label
  {0pt}                       % separation
  {}                          % before-code
  [\hrule]  % after-code

% \titlespacing*{⟨command ⟩}{⟨left⟩}{⟨before-sep⟩}{⟨after-sep⟩}[⟨right-sep⟩]
\titlespacing*{\section}
  {0pt}
  {0pt}
  {0pt}
  [0pt]

\titleformat{\subsection}
  {\normalfont\bfseries\color{mygreen}} % format applied to whole title
  {}                          % label
  {0pt}                       % separation
  {}                          % before-code
  []                          % after-code

% \titlespacing*{⟨command ⟩}{⟨left⟩}{⟨before-sep⟩}{⟨after-sep⟩}[⟨right-sep⟩]
\titlespacing*{\subsection}
  {0pt}
  {0pt}
  {0pt}
  [0pt]

\newcommand{\subtitle}[3]{{\vspace{3pt}\bfseries #1, }{\bfseries \color{mygreen}#2}\hfill #3 \\}
