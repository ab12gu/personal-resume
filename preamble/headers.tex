\usepackage{titlesec}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Custom section format
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\titleformat{\section}
  {\normalfont\Large\bfseries} % format applied to whole title
  {}                          % label
  {0pt}                       % separation
  {\vspace{-0.5em}}                          % before-code
  [\hrule]  % after-code

\titleformat{\subsection}
  {\normalfont\bfseries} % format applied to whole title
  {}                          % label
  {0pt}                       % separation
  {\vspace{-0.5em}}                          % before-codeA
  []  % after-code

\newcommand{\subtitle}[3]{{\bfseries #1, }{\bfseries \color{mygreen}#2}\hfill #3}
